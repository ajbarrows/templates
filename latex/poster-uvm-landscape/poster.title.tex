% Simple title storage - formatting handled in maketitle
\makeatletter
\def\title#1{\gdef\@title{#1}}
\makeatother



\makeatletter
\newcommand\insertlogoi[2][]{\def\@insertlogoi{\includegraphics[#1]{#2}}}
\newcommand\insertlogoii[2][]{\def\@insertlogoii{\includegraphics[#1]{#2}}}
% Calculate offset to center logos between title text and poster edge
\newlength\LogoSep
\setlength\LogoSep{0.1\dimexpr\TP@visibletextwidth-\TP@titlewidth\relax}
%
\insertlogoi[width=8cm]{abcd}
\insertlogoii[width=8cm]{complexbrain}
%
\renewcommand\maketitle[1][]{  % #1 keys
	\normalsize
	\setkeys{title}{#1}
	% Title dummy to get title height
	\node[transparent,inner sep=\TP@titleinnersep, line width=\TP@titlelinewidth, anchor=north, minimum width=\TP@visibletextwidth-2\TP@titleinnersep, text width=0.62\TP@visibletextwidth, align=center]
	(TP@title) at ($(0, 0.5\textheight-\TP@titletotopverticalspace)$) {{\fontsize{55}{66}\selectfont\sloppy\@title}\\[0.5em]{\@author}\\[0.3em]{\@institute}};
	\draw let \p1 = ($(TP@title.north)-(TP@title.south)$) in node {
		\setlength{\TP@titleheight}{\y1}
		\setlength{\titleheight}{\y1}
		\global\TP@titleheight=\TP@titleheight
		\global\titleheight=\titleheight
	};

	% Compute title position
	\setlength{\titleposleft}{-0.5\titlewidth}
	\setlength{\titleposright}{\titleposleft+\titlewidth}
	\setlength{\titlepostop}{0.5\textheight-\TP@titletotopverticalspace}
	\setlength{\titleposbottom}{\titlepostop-\titleheight}

	% Title style (background)
	\TP@titlestyle

	% Title node - directly construct content for proper wrapping
	\node[inner sep=\TP@titleinnersep, line width=\TP@titlelinewidth, anchor=north, minimum width=\TP@visibletextwidth-2\TP@titleinnersep, text width=0.62\TP@visibletextwidth, align=center]
	at (0,0.5\textheight-\TP@titletotopverticalspace)
	(title)
	{\color{titlefgcolor}{\fontsize{55}{66}\selectfont\sloppy\@title}\\[0.5em]{\@author}\\[0.3em]{\@institute}};
	
	\node[inner sep=0pt,anchor=center]
	at ([xshift=\LogoSep]title.west)
	{\@insertlogoi};

	\node[inner sep=0pt,anchor=center]
	at ([xshift=-\LogoSep]title.east)
	{\@insertlogoii};
	
	% Settings for blocks
	\normalsize
	\setlength{\TP@blocktop}{\titleposbottom-\TP@titletoblockverticalspace}
}
\makeatother
